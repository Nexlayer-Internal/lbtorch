cmake_minimum_required(VERSION 3.18)
project(nemotron_h_check CXX)
# Standalone CPU build of torch::models::nemotron_h + its gradient gate against the prebuilt libtorch (same pattern as the trainer app's qwen35 target):
# compiles the model sources of THIS clone and links ONLY prebuilt libc10 / libtorch_cpu / libtorch_cuda (+libc10_cuda). Nothing here touches a GPU.
get_filename_component(LBT_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LBT_BUILD "/mnt/raid/lbtorch-checkpoint-build/src/build" CACHE PATH "prebuilt libtorch build tree (read-only: libs + generated headers)")
set(LBT_TREE  "/mnt/raid/lbtorch-checkpoint-build/src"       CACHE PATH "same tree, for third_party headers absent from the clone (nlohmann)")
set(CUDA_INC  "/usr/local/cuda-13.0/include"                 CACHE PATH "")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
add_library(nemotron_h STATIC
  ${LBT_SRC}/torch/csrc/api/src/utils/safetensors.cpp
  ${LBT_SRC}/torch/csrc/api/src/models/nemotron_h.cpp)
set(OVL ${CMAKE_BINARY_DIR}/overlay_include)
file(GLOB_RECURSE NEWH RELATIVE ${LBT_SRC}/torch/csrc/api/include ${LBT_SRC}/torch/csrc/api/include/torch/models/*.h ${LBT_SRC}/torch/csrc/api/include/torch/utils/safetensors.h)
foreach(H ${NEWH})
  if(NOT EXISTS ${LBT_TREE}/torch/csrc/api/include/${H})
    get_filename_component(D ${OVL}/${H} DIRECTORY)
    file(MAKE_DIRECTORY ${D})
    file(CREATE_LINK ${LBT_SRC}/torch/csrc/api/include/${H} ${OVL}/${H} SYMBOLIC)
  endif()
endforeach()
target_include_directories(nemotron_h SYSTEM PUBLIC ${OVL}
  ${LBT_TREE}/torch/csrc/api/include ${LBT_TREE}/torch/csrc/api ${LBT_TREE}/torch/csrc ${LBT_TREE}/torch/headeronly ${LBT_TREE}
  ${LBT_TREE}/aten/src ${LBT_BUILD}/aten/src ${LBT_BUILD})
target_include_directories(nemotron_h SYSTEM PUBLIC ${LBT_TREE}/third_party/nlohmann/include ${CUDA_INC} ${LBT_BUILD}/include)
target_compile_definitions(nemotron_h PUBLIC USE_CUDA)
target_compile_options(nemotron_h PRIVATE -fPIC -Wno-deprecated-declarations)
foreach(L c10 c10_cuda torch_cpu torch_cuda)
  add_library(lbt_${L} SHARED IMPORTED)
  set_target_properties(lbt_${L} PROPERTIES IMPORTED_LOCATION ${LBT_BUILD}/lib/lib${L}.so)
endforeach()
target_link_libraries(nemotron_h PUBLIC -Wl,--no-as-needed lbt_torch_cuda lbt_torch_cpu lbt_c10_cuda lbt_c10 -Wl,--as-needed)
set(CMAKE_BUILD_RPATH ${LBT_BUILD}/lib)
add_executable(nemotron_h_check nemotron_h_check.cpp)
target_compile_options(nemotron_h_check PRIVATE -Wall -Wextra -Wno-deprecated-declarations -Wno-unused-parameter)
target_link_libraries(nemotron_h_check nemotron_h)
